INCLUDE Irvine32.inc

.data
    Source BYTE "Hello Assembly!", 0
    Target BYTE 16 DUP(?)
    msgSource BYTE "Source String:   ", 0
    msgTarget BYTE "Target String: ", 0

.code
main PROC
    mov ecx, 15             
    mov esi, 0              

CopyLoop:
    mov al, Source[esi]     
    mov Target[esi], al    
    inc esi
    loop CopyLoop

    mov Target[esi], 0      
    mov edx, OFFSET msgSource
    call WriteString
    mov edx, OFFSET Source
    call WriteString
    call Crlf
    mov edx, OFFSET msgTarget
    call WriteString
    mov edx, OFFSET Target
    call WriteString
    call Crlf
    exit
main ENDP
END main
