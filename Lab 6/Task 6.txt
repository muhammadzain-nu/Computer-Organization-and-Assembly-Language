INCLUDE Irvine32.inc

.data
    msg1 BYTE "Message 1", 0dh, 0ah, 0
    msg2 BYTE "Message 2", 0dh, 0ah, 0

.code
main PROC
    mov edx, OFFSET msg1
    call WriteString

    jmp SkipSecondMsg

    mov edx, OFFSET msg2
    call WriteString

SkipSecondMsg:
    exit
main ENDP
END main
