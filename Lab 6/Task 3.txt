INCLUDE Irvine32.inc

.data
    msgWelcome BYTE "Welcome Message", 0dh, 0ah, 0
    msgAd      BYTE "Advertisement Section", 0dh, 0ah, 0
    msgBilling BYTE "Billing Notice", 0dh, 0ah, 0

.code
main PROC
    mov edx, OFFSET msgWelcome
    call WriteString

    jmp SkipAd

    mov edx, OFFSET msgAd
    call WriteString

SkipAd:
    mov edx, OFFSET msgBilling
    call WriteString

    exit
main ENDP
END main
