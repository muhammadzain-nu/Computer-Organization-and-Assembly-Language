INCLUDE Irvine32.inc

.data
    name1 BYTE 50 DUP(?)
    dept1 BYTE 50 DUP(?)
    name2 BYTE 50 DUP(?)
    dept2 BYTE 50 DUP(?)
    name3 BYTE 50 DUP(?)
    dept3 BYTE 50 DUP(?)
    
    spaceChar BYTE " ", 0
    header BYTE "Student Records:", 0dh, 0ah, 0

.code
main PROC
    mov edx, OFFSET name1
    mov ecx, SIZEOF name1 - 1
    call ReadString

    mov edx, OFFSET dept1
    mov ecx, SIZEOF dept1 - 1
    call ReadString

    mov edx, OFFSET name2
    mov ecx, SIZEOF name2 - 1
    call ReadString

    mov edx, OFFSET dept2
    mov ecx, SIZEOF dept2 - 1
    call ReadString

    mov edx, OFFSET name3
    mov ecx, SIZEOF name3 - 1
    call ReadString

    mov edx, OFFSET dept3
    mov ecx, SIZEOF dept3 - 1
    call ReadString

    mov edx, OFFSET header
    call WriteString

    mov edx, OFFSET name1
    call WriteString
    mov edx, OFFSET spaceChar
    call WriteString
    mov edx, OFFSET dept1
    call WriteString
    call Crlf

    mov edx, OFFSET name2
    call WriteString
    mov edx, OFFSET spaceChar
    call WriteString
    mov edx, OFFSET dept2
    call WriteString
    call Crlf

    mov edx, OFFSET name3
    call WriteString
    mov edx, OFFSET spaceChar
    call WriteString
    mov edx, OFFSET dept3
    call WriteString
    call Crlf

    exit
main ENDP
END main
