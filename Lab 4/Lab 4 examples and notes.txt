MOV: to move the value from source to destination
Add: to add value from source to destination
Sub: to sub "                "
INC: adds 1
DEC: subtracts 1
MOVZX: moves unsigned smaller registers into Large registers
MOVSX: moves signed smaller register into Large registers
XCHG: Swaps the value
NEG: Takes 2's compliment of the value.
Equal Sign: assigns the value 




.data
oneByte BYTE 78h
oneWord WORD 1234h
oneDword DWORD 12345678h
.code
Mov eax, 0 ; EAX = 00000000h
Mov al, oneByte ; EAX = 00000078h
Mov ax, oneWord ; EAX = 00001234h
Mov eax, oneDword ; EAX = 12345678h
Mov ax, 0 ; EAX = 12340000h
